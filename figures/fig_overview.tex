\begin{tikzpicture}[font=\scriptsize]

\node[hplain,anchor=west,text width=23mm] (raw) at (0,0.60)

\node[hplain,anchor=west,text width=23mm] (text) at (0,-0.65)
  {per-sensor text\\ \tiny ``a smartwatch accelerometer on the wrist''};

\node[hbox,anchor=west,text width=27mm,minimum height=13mm] (tok) at (2.75,0)

\node[hbox,anchor=west,text width=25mm,minimum height=13mm] (enc) at (5.95,0)
  {\textbf{set encoder}\\ \tiny RoPE over physical time;\\

\node[hbox2,anchor=north,text width=48mm] (obj) at (5.60,-0.92)

\node[hbox2,anchor=west,text width=27mm,minimum height=13mm] (dec) at (10.10,0)

\node[hplain,anchor=west,text width=22mm] (out) at (13.60,0)
  {evidence per candidate\\ $+$ calibrated rejection\\

   \tiny encoded exemplars, $k{=}0$ allowed\\

\draw[harrow] (cand.north) -- ($(dec.south)+(7mm,0)$);
